name: F-004 v3 KMI con el define corregido

# Tres etapas. La tercera existe porque stgdiff SI existe: es un binario listo en
# kernel/prebuilts/build-tools/linux-x86/bin/stgdiff. Es linux-x86, asi que NO corre
# en arm64: se COMPILA en arm64 y se COMPARA en x86. stg LEE el ELF, no lo ejecuta.
on:
  push:
    branches: ['titan/f-004-kmi']
    paths: ['.github/workflows/f-004-kmi-v3.yml']
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    name: v3-build-${{ matrix.brazo }}
    runs-on: ubuntu-24.04-arm
    timeout-minutes: 350
    strategy:
      fail-fast: false
      matrix:
        brazo: [baseline, siao]
    steps:
      - uses: actions/checkout@v4

      - name: Identificar la maquina, verbatim
        run: |
          set -x
          uname -a
          nproc
          free -m
          df -h /

      - name: md5 del instrumento
        run: md5sum tools/f004_kmi_v3.py

      - name: Herramientas de kernel
        run: |
          sudo apt-get update -qq
          sudo apt-get install -y -qq --no-install-recommends clang lld llvm bc bison flex libssl-dev libelf-dev python3 rsync kmod cpio zstd
          clang --version | head -2
          ld.lld --version | head -1
          make --version | head -1
          openssl version

      - name: Liberar disco
        run: |
          df -h /
          sudo rm -rf /usr/share/dotnet /usr/local/lib/android /opt/ghc || true
          df -h /

      - name: Compilar ${{ matrix.brazo }}
        env:
          F004_OUT: mediciones/f-004-v3
          F004_WORK: /tmp/f004v3
          F004_RAMA: android15-6.6
        run: python3 tools/f004_kmi_v3.py build ${{ matrix.brazo }}

      - name: Subir evidencia y symvers
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: f004v3-${{ matrix.brazo }}
          path: |
            mediciones/f-004-v3/f004v3-*
            mediciones/f-004-v3/symvers-*
            mediciones/f-004-v3/config-*
          retention-days: 5
          if-no-files-found: warn

      - name: Subir el ELF para el stgdiff x86
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: f004v3-elf-${{ matrix.brazo }}
          path: mediciones/f-004-v3/elf-${{ matrix.brazo }}.tar.zst
          retention-days: 5
          if-no-files-found: warn

  criba:
    name: v3-criba-symvers
    needs: build
    if: always()
    runs-on: ubuntu-24.04-arm
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/download-artifact@v4
        with:
          pattern: 'f004v3-*'
          path: /tmp/art
      - name: Juntar
        run: |
          mkdir -p mediciones/f-004-v3
          find /tmp/art -type f | sort
          cp -v /tmp/art/f004v3-baseline/* mediciones/f-004-v3/ 2>/dev/null || true
          cp -v /tmp/art/f004v3-siao/* mediciones/f-004-v3/ 2>/dev/null || true
          ls -la mediciones/f-004-v3/
      - name: Criba por Module.symvers
        env:
          F004_OUT: mediciones/f-004-v3
          F004_RAMA: android15-6.6
        run: python3 tools/f004_kmi_v3.py comparar || true
      - name: Commitear
        if: always()
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add -A mediciones
          git diff --cached --quiet && echo 'sin cambios' && exit 0
          git commit -m "evidencia(f-004 v3): criba de Module.symvers (run ${{ github.run_id }})"
          for i in 1 2 3 4 5; do
            git pull --rebase --autostash origin "${{ github.ref_name }}" && git push origin HEAD:"${{ github.ref_name }}" && exit 0
            sleep $((i * 5))
          done
          exit 1

  stgdiff:
    name: v3-stgdiff-x86
    needs: build
    if: always()
    runs-on: ubuntu-24.04
    timeout-minutes: 90
    steps:
      - uses: actions/checkout@v4

      - name: Identificar la maquina
        run: |
          set -x
          uname -m
          nproc
          df -h /

      - name: Bajar stg y stgdiff del prebuilt de Google, con sha256 y su help
        run: |
          set -x
          mkdir -p /tmp/stg
          base=https://android.googlesource.com/kernel/prebuilts/build-tools/+/refs/heads/main/linux-x86/bin
          curl -sSL "$base/stg?format=TEXT" | base64 -d > /tmp/stg/stg
          curl -sSL "$base/stgdiff?format=TEXT" | base64 -d > /tmp/stg/stgdiff
          chmod +x /tmp/stg/stg /tmp/stg/stgdiff
          ls -la /tmp/stg
          sha256sum /tmp/stg/stg /tmp/stg/stgdiff
          file /tmp/stg/stg /tmp/stg/stgdiff
          /tmp/stg/stg --help 2>&1 | head -60 || true
          /tmp/stg/stgdiff --help 2>&1 | head -60 || true

      - uses: actions/download-artifact@v4
        with:
          pattern: 'f004v3-elf-*'
          path: /tmp/elf

      - name: Desempaquetar los dos ELF
        run: |
          set -x
          find /tmp/elf -type f | xargs -r ls -la
          for b in baseline siao; do
            mkdir -p /tmp/k/$b
            f=$(find /tmp/elf -name elf-$b.tar.zst | head -1)
            if [ -n "$f" ]; then tar -x --zstd -f "$f" -C /tmp/k/$b || true; fi
            ls -la /tmp/k/$b | head -5 || true
          done

      - name: Generar los .stg y correr stgdiff
        run: python3 tools/f004_stg_x86.py

      - name: Commitear
        if: always()
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add -A mediciones
          git diff --cached --quiet && echo 'sin cambios' && exit 0
          git commit -m "evidencia(f-004 v3): stgdiff real con el prebuilt x86 (run ${{ github.run_id }})"
          for i in 1 2 3 4 5; do
            git pull --rebase --autostash origin "${{ github.ref_name }}" && git push origin HEAD:"${{ github.ref_name }}" && exit 0
            sleep $((i * 7))
          done
          exit 1

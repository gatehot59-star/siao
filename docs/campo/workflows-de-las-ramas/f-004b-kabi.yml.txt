name: F-004b KABI del padding

# Dos brazos y una comparacion, todo en un run:
#   baseline -> arbol LIMPIO + gki_defconfig puro     (la referencia)
#   kabi     -> arbol PARCHEADO + gki_defconfig + fragmento
# La unica variable es el fragmento MAS el parche que lo hace caber en el padding.
# El stgdiff corre en x86 porque el binario de Google es linux-x86.
on:
  push:
    branches: ['titan/f-004-kmi']
    paths:
      - 'tools/f004b_kabi.py'
      - '.github/workflows/f-004b-kabi.yml'
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    name: f004b-${{ matrix.brazo }}
    runs-on: ubuntu-24.04-arm
    timeout-minutes: 350
    strategy:
      fail-fast: false
      matrix:
        brazo: [baseline, kabi]
    steps:
      - uses: actions/checkout@v4
      - name: Identificar la maquina, verbatim
        run: |
          set -x
          uname -a
          nproc
          free -m
          df -h /
      - name: md5 del instrumento
        run: md5sum tools/f004b_kabi.py
      - name: Herramientas de kernel
        run: |
          sudo apt-get update -qq
          sudo apt-get install -y -qq --no-install-recommends clang lld llvm bc bison flex libssl-dev libelf-dev python3 rsync kmod cpio zstd
          clang --version | head -2
          openssl version
      - name: Liberar disco
        run: |
          df -h /
          sudo rm -rf /usr/share/dotnet /usr/local/lib/android /opt/ghc || true
          df -h /
      - name: Compilar ${{ matrix.brazo }}
        env:
          F004_OUT: mediciones/f-004b
          F004_WORK: /tmp/f004b
          F004_RAMA: android15-6.6
        run: python3 tools/f004b_kabi.py build ${{ matrix.brazo }}
      - name: Subir evidencia
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: f004b-${{ matrix.brazo }}
          path: |
            mediciones/f-004b/f004b-*
            mediciones/f-004b/symvers-*
            mediciones/f-004b/config-*
          retention-days: 5
          if-no-files-found: warn
      - name: Subir el ELF
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: f004b-elf-${{ matrix.brazo }}
          path: mediciones/f-004b/elf-${{ matrix.brazo }}.tar.zst
          retention-days: 5
          if-no-files-found: warn

  stgdiff:
    name: f004b-stgdiff-x86
    needs: build
    if: always()
    runs-on: ubuntu-24.04
    timeout-minutes: 90
    steps:
      - uses: actions/checkout@v4
      - name: Bajar los ELF de este mismo run
        uses: actions/download-artifact@v4
        with:
          pattern: 'f004b-elf-*'
          path: /tmp/elf
      - name: Bajar la evidencia de los brazos
        uses: actions/download-artifact@v4
        with:
          pattern: 'f004b-baseline'
          path: /tmp/art
        continue-on-error: true
      - name: Desempaquetar
        run: |
          set -x
          find /tmp/elf -type f | xargs -r ls -la
          for b in baseline kabi; do
            mkdir -p /tmp/k/$b
            f=$(find /tmp/elf -name elf-$b.tar.zst | head -1)
            if [ -n "$f" ]; then tar -x --zstd -f "$f" -C /tmp/k/$b || true; fi
            ls -la /tmp/k/$b | head -4 || true
          done
      - name: stg + stgdiff, las dos pasadas
        run: python3 tools/f004b_stg.py
      - name: Commitear
        if: always()
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add -A mediciones
          git diff --cached --quiet && echo 'sin cambios' && exit 0
          git commit -m "evidencia(F-004b): stgdiff del parche KABI (run ${{ github.run_id }})"
          for i in 1 2 3 4 5; do
            git pull --rebase --autostash origin titan/f-004-kmi && git push origin HEAD:titan/f-004-kmi && exit 0
            sleep $((i * 7))
          done
          exit 1
      - name: Resumen
        if: always()
        run: head -80 mediciones/f-004b/F-004b-VEREDICTO.txt >> "$GITHUB_STEP_SUMMARY" || true

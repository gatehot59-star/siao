name: F-004d en android14-6.1

# VERSION 3. La v2 (run 34126598730) probo el paso serial, pero mi arreglo estaba
# incompleto: archscripts corrio SIN HOSTCFLAGS, tardo 0,0 s y no hizo trabajo;
# despues el build con HOSTCFLAGS invalido fixdep y lo recompilo en paralelo.
# Esta v3 unifica HOSTCFLAGS en los TRES pasos, y guarda segundos + presencia de
# HOSTCC para probar que el paso realmente mitigo algo.
#
# v1: pahole faltaba, fixdep 126, tar sin vmlinux.
# v2: pahole resuelto, guard de tar resuelto, pero HOSTCFLAGS no unificado.
# v3: la firma definitiva de host esta en gki_defconfig, archscripts y build.
on:
  push:
    branches: ['titan/f-004-kmi']
    paths:
      - '.github/workflows/f-004d-61.yml'
      - 'tools/f004d61_build.py'
  workflow_dispatch:
permissions:
  contents: write
jobs:
  build:
    name: f004d61v3-${{ matrix.brazo }}
    runs-on: ubuntu-24.04-arm
    timeout-minutes: 350
    strategy:
      fail-fast: false
      matrix:
        brazo: [baseline, ocho]
    steps:
      - uses: actions/checkout@v4
      - name: Identificar la maquina, verbatim
        run: |
          set -x
          uname -a
          nproc
          free -m
          df -h /
      - name: md5 del instrumento unico
        run: md5sum tools/f004d61_build.py
      - name: Herramientas de kernel con pahole
        run: |
          sudo apt-get update -qq
          sudo apt-get install -y -qq --no-install-recommends clang lld llvm bc bison flex libssl-dev libelf-dev python3 rsync kmod cpio zstd dwarves
          echo "pahole -> $(command -v pahole || echo AUSENTE)"
          pahole --version 2>&1 | head -1 || true
      - name: Liberar disco
        run: |
          df -h /
          sudo rm -rf /usr/share/dotnet /usr/local/lib/android /opt/ghc || true
          df -h /
      - name: Compilar brazo
        env:
          F004_OUT: mediciones/f-004d-61
          F004_WORK: /tmp/f004d61
          F004_RAMA: android14-6.1
        run: python3 tools/f004d61_build.py ${{ matrix.brazo }}
      - name: Subir evidencia
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: f004d61v3-${{ matrix.brazo }}
          path: mediciones/f-004d-61/
          retention-days: 5
          if-no-files-found: warn
      - name: Commitear evidencia del brazo
        if: always()
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          git add -A mediciones/f-004d-61
          git diff --cached --quiet && exit 0
          git commit -m "evidencia(F-004d@6.1 v3): brazo ${{ matrix.brazo }} (run ${{ github.run_id }})"
          for i in 1 2 3 4 5; do
            git pull --rebase --autostash origin titan/f-004-kmi && git push origin HEAD:titan/f-004-kmi && exit 0
            sleep $((i * 8))
          done
          exit 1
  comparar:
    name: f004d61v3-stgdiff-x86
    needs: build
    if: always()
    runs-on: ubuntu-24.04
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4
        with:
          ref: titan/f-004-kmi
      - uses: actions/download-artifact@v4
        with:
          pattern: 'f004d61v3-*'
          path: /tmp/art
      - name: R-11, copiar solo la evidencia de cada brazo
        run: |
          set -x
          mkdir -p mediciones/f-004d-61
          find /tmp/art -type f | sort
          # cada artifact tiene una copia del mismo OUT; no pisar con glob ciego
          cp -rv /tmp/art/f004d61v3-baseline/* mediciones/f-004d-61/ 2>/dev/null || true
          cp -rv /tmp/art/f004d61v3-ocho/* mediciones/f-004d-61/ 2>/dev/null || true
          ls -la mediciones/f-004d-61/
      - name: Desempaquetar y verificar vmlinux adentro
        run: |
          set -x
          for b in baseline ocho; do
            mkdir -p /tmp/k/$b
            f=$(find /tmp/art -name "elf-$b-android14-6.1.tar.zst" | head -1)
            if [ -z "$f" ]; then echo "NO llego tarball $b"; continue; fi
            tar -t --zstd -f "$f" | head -3
            tar -x --zstd -f "$f" -C /tmp/k/$b || true
            test -f /tmp/k/$b/vmlinux || echo "SIN vmlinux $b"
          done
      - name: stg y stgdiff
        env:
          F004_OUT: mediciones/f-004d-61
          F004_RAMA: android14-6.1
        run: python3 tools/f004d61_stg.py
      - name: Commitear
        if: always()
        run: |
          git config user.name 'github-actions[bot]'
          git config user.email '41898282+github-actions[bot]@users.noreply.github.com'
          rm -f mediciones/f-004d-61/elf-*.tar.zst
          git add -A mediciones/f-004d-61
          git diff --cached --quiet && exit 0
          git commit -m "evidencia(F-004d@6.1 v3): stgdiff (run ${{ github.run_id }})"
          for i in 1 2 3 4 5; do
            git pull --rebase --autostash origin titan/f-004-kmi && git push origin HEAD:titan/f-004-kmi && exit 0
            sleep $((i * 8))
          done
          exit 1
      - name: Resumen
        if: always()
        run: tail -60 mediciones/f-004d-61/F-004d-61-VEREDICTO.txt >> "$GITHUB_STEP_SUMMARY" || true
